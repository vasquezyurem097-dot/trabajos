#include <stdio.h>

int main() {
    float fahrenheit, celsius;
    char respuesta;

    printf("Dame la temperatura en Fahrenheit: ");
    scanf("%f", &fahrenheit);

    celsius = (fahrenheit - 32) * 5 / 9;

    printf("La temperatura en Celsius es: %.2f\n", celsius);

    if (celsius > 31) {
        printf("La temperatura esta pasando los 31 grados Celsius.\n");
    }
    else if (celsius <= 31) {
        printf("La temperatura no esta pasando los 31 grados Celsius.\n");
    }

    return 0;
}
